def solve_n_queens(n):
    board = [-1] * n

    def is_safe(row, col):
        for i in range(row):
            if board[i] == col or abs(board[i] - col) == abs(i - row):
                return False
        return True

    def solve(row):
        if row == n:
            for i in range(n):
                for j in range(n):
                    print("Q" if board[i] == j else ".", end=" ")
                print()
            return True

        for col in range(n):
            if is_safe(row, col):
                board[row] = col
                if solve(row + 1):
                    return True
                board[row] = -1
        return False

    if not solve(0):
        print("No solution exists")

n = int(input("Enter number of queens: "))
solve_n_queens(n)

# Distance Vector Routing Algorithm

n = int(input("Enter number of routers: "))

print("Enter the cost matrix (use 999 for infinity):")
cost = []

for i in range(n):
    cost.append(list(map(int, input().split())))

dist = [row[:] for row in cost]

for k in range(n):
    for i in range(n):
        for j in range(n):
            if dist[i][j] > dist[i][k] + dist[k][j]:
                dist[i][j] = dist[i][k] + dist[k][j]

print("\nShortest distance between routers:")
for i in range(n):
    print("Router", i + 1, ":", dist[i])

import socket
import time

s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
s.settimeout(2)

for i in range(4):
    start = time.time()

    s.sendto("Ping".encode(), ('localhost', 12345))

    try:
        data, addr = s.recvfrom(1024)
        end = time.time()

        print("Reply:", data.decode(),
              "Time:", round((end - start) * 1000, 2), "ms")
    except socket.timeout:
        print("Request timed out")

s.close()

file transfer in client and server 

import socket

s = socket.socket()
s.bind(('localhost', 12345))
s.listen(1)

print("Server waiting for connection...")
conn, addr = s.accept()
print("Client connected:", addr)

filename = conn.recv(1024).decode()

with open("received_" + filename, "wb") as f:
    while True:
        data = conn.recv(1024)
        if not data:
            break
        f.write(data)

print("File received successfully")
conn.close()
s.close()

import socket
import os

s = socket.socket()
s.connect(('localhost', 12345))

filename = input("Enter filename to send: ")

s.send(filename.encode())

with open(filename, "rb") as f:
    while True:
        data = f.read(1024)
        if not data:
            break
        s.sendall(data)

s.shutdown(socket.SHUT_WR)
print("File sent successfully")

s.close()
