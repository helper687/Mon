n = int(input("Enter number of queens: "))
board = []

def solve(row):
    if row == n:
        print("Solution:", board)
        return True

    for col in range(n):
        if all(col != board[i] and
               abs(col - board[i]) != row - i
               for i in range(row)):
            board.append(col)

            if solve(row + 1):
                return True

            board.pop()

    return False

if not solve(0):
    print("No solution exists")

Explanation
Backtracking places queens row by row on the chessboard. The board list stores each queen's column position, while the solve(row) function tries to place a queen in a safe position. The all() function checks column and diagonal conflicts, and abs() calculates the difference between positions. If a position is unsafe, board.pop() removes the last queen and tries another position. If no valid arrangement exists, the program prints No solution exists.

Result: The N-Queen problem is solved using backtracking.

visited = {start}

while q:
    state, path = q.popleft()

    if state == goal:
        print("Solution:")
        for s in path + [state]:
            print(s)
        break

    z = state.index(0)

    for m in [-n, n, -1, 1]:
        p = z + m

        if 0 <= p < size and not (
            (m == -1 and z % n == 0) or
            (m == 1 and z % n == n-1)
        ):
            new = list(state)
            new[z], new[p] = new[p], new[z]
            new = tuple(new)

            if new not in visited:
                visited.add(new)
                q.append((new, path + [state]))
else:
    print("No solution found")

def alphabeta(depth, index, maximizing, values, alpha, beta):
    if depth == 2:
        return values[index]

    if maximizing:
        best = -999

        for i in range(2):
            value = alphabeta(depth + 1, index * 2 + i,
                              False, values, alpha, beta)
            best = max(best, value)
            alpha = max(alpha, best)

            if beta <= alpha:
                break

        return best

    else:
        best = 999

        for i in range(2):
            value = alphabeta(depth + 1, index * 2 + i,
                              True, values, alpha, beta)
            best = min(best, value)
            beta = min(beta, best)

            if beta <= alpha:
                break

        return best


values = list(map(int, input("Enter 4 leaf values: ").split()))

answer = alphabeta(0, 0, True, values, -999, 999)
print("Optimal value:", answer)

Optimal value: 5/6

Explanation
Alpha-Beta Pruning improves the Minimax algorithm by avoiding branches that cannot affect the final decision. The maximizing player chooses the maximum value, while the minimizing player chooses the minimum value. alpha stores the best value for the maximizing player, and beta stores the best value for the minimizing player. When beta <= alpha, the remaining branches are pruned because they cannot change the result.

Result: Alpha-Beta pruning is implemented to find the optimal value while reducing unnecessary searches.






# AND gate using Perceptron Learning Rule

inputs = [(0, 0), (0, 1), (1, 0), (1, 1)]
targets = [0, 0, 0, 1]

w1 = 0
w2 = 0
bias = 0
lr = 0.1

for epoch in range(10):
    for i in range(4):
        x1, x2 = inputs[i]
        output = 1 if w1*x1 + w2*x2 + bias > 0 else 0
        error = targets[i] - output

        w1 += lr * error * x1
        w2 += lr * error * x2
        bias += lr * error

print("Predictions:")
for x1, x2 in inputs:
    output = 1 if w1*x1 + w2*x2 + bias > 0 else 0
    print(x1, x2, "=", output)

import random

# Fitness function
def fitness(x):
    return x * x

# Initial population
population = [random.randint(0, 10) for _ in range(4)]

for i in range(10):
    population.sort(key=fitness, reverse=True)

    # Selection
    parent1 = population[0]
    parent2 = population[1]

    # Crossover
    child = (parent1 + parent2) // 2

    # Mutation
    if random.random() < 0.3:
        child = random.randint(0, 10)

    population[-1] = child

best = max(population, key=fitness)

print("Best solution:", best)
print("Maximum fitness:", fitness(best))

Best solution: 10
Maximum fitness: 100


Explanation
A Genetic Algorithm (GA) is an evolutionary optimization technique inspired by natural selection. The population contains possible solutions, and the fitness function evaluates each solution. Selection chooses the best parents, crossover combines them to produce a child, and mutation introduces random changes to maintain diversity. These steps repeat for several generations to find a good solution.

Result: A Genetic Algorithm is implemented to find a solution with high fitness.
